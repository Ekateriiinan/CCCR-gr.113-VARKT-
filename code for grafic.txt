import numpy as np
import matplotlib.pyplot as plt

# Константы
g0 = 9.81  # Ускорение свободного падения, м/с^2
R = 6371000  # Радиус Земли, м
rho0 = 1.225  # Плотность воздуха на уровне моря, кг/м^3
Cd = 0.5  # Коэффициент лобового сопротивления
A = 10.0  # Площадь поперечного сечения ракеты, м^2
ve = 2800  # Скорость истечения газов, м/с
m0 = 120  # Начальная масса ракеты, т
burn_time = 150  # Время работы двигателя, с
thrust = 2000  # Тяга двигателя, кН
T = 450  # Общее время моделирования, с
dt = 1  # Шаг времени, с

# Ступени ракеты (время сброса и массы)
stages = [(0, 120), (60, 90), (150, 40), (300, 20)]  # (время, масса)

# Функция плотности воздуха в зависимости от высоты
def air_density(h):
    return rho0 * np.exp(-h / 8400)

# Система уравнений: dV/dt, dH/dt
def rocket_dynamics(t, V, H, M):
    if t < burn_time:
        F_thrust = thrust * 1000  # Перевод в Ньютоны
    else:
        F_thrust = 0  # Двигатели выключены

    g = g0 * (R / (R + H))**2  # Ускорение свободного падения
    rho = air_density(H)  # Плотность воздуха
    F_drag = 0.5 * Cd * A * rho * V**2  # Сила сопротивления воздуха

    dV_dt = (F_thrust - F_drag - M * 1000 * g) / (M * 1000)  # Изменение скорости
    dH_dt = V  # Изменение высоты
    return dV_dt, dH_dt

# Метод Рунге-Кутты 4-го порядка
time = np.arange(0, T + dt, dt)
V, H = [0], [0]  # Начальные значения скорости и высоты
M = [m0]  # Начальная масса
current_stage = 0

for t in time[:-1]:
    # Проверка на сброс ступени
    if current_stage < len(stages) - 1 and t >= stages[current_stage + 1][0]:
        current_stage += 1
        M[-1] = stages[current_stage][1]  # Масса изменяется дискретно
    
    k1_v, k1_h = rocket_dynamics(t, V[-1], H[-1], M[-1])
    k2_v, k2_h = rocket_dynamics(t + dt/2, V[-1] + dt*k1_v/2, H[-1] + dt*k1_h/2, M[-1])
    k3_v, k3_h = rocket_dynamics(t + dt/2, V[-1] + dt*k2_v/2, H[-1] + dt*k2_h/2, M[-1])
    k4_v, k4_h = rocket_dynamics(t + dt, V[-1] + dt*k3_v, H[-1] + dt*k3_h, M[-1])
    
    V.append(V[-1] + dt * (k1_v + 2*k2_v + 2*k3_v + k4_v) / 6)
    H.append(H[-1] + dt * (k1_h + 2*k2_h + 2*k3_h + k4_h) / 6)
    M.append(M[-1])  # Масса остаётся постоянной между сбросами

# Данные для графика скорости
time_velocity = np.array([0, 50, 120, 200, 300, 400])  # Время, с
velocity = np.array([0, 500, 1600, 1500, 2000, 2000])  # Скорость, м/с

# Линейная интерполяция для более гладкой кривой
time_fine = np.linspace(0, 400, 500)  # Подробная сетка времени для плавного графика
velocity_fine = np.interp(time_fine, time_velocity, velocity)

# Построение всех трёх графиков
plt.figure(figsize=(12, 12))

# График высоты
plt.subplot(3, 1, 1)
plt.plot(time, H, 'b', linewidth=2, label="Высота над уровнем моря")
plt.title("График изменения высоты")
plt.ylabel("Высота, м")
plt.xlabel("Время, с")
plt.legend()
plt.grid(True)

# График скорости
plt.subplot(3, 1, 2)
plt.plot(time_fine, velocity_fine, 'r', linewidth=2, label="Скорость на поверхности")
plt.title("График изменения скорости")
plt.ylabel("Скорость, м/с")
plt.xlabel("Время, с")
plt.legend()
plt.grid(True)

# График массы
plt.subplot(3, 1, 3)
plt.step(time, M, 'g', linewidth=2, where='post', label="Масса ракеты")
plt.title("График изменения массы ракеты")
plt.ylabel("Масса, т")
plt.xlabel("Время, с")
plt.legend()
plt.grid(True)

plt.tight_layout()
plt.show()
